Requetes SQL avec jointures utilisees

-- Evenements par type de menace (INNER JOIN)

SELECT tt.name, tt.severity_level, COUNT(se.id)
FROM threat_types tt
LEFT JOIN security_events se ON tt.id = se.threat_type_id
GROUP BY tt.id, tt.name, tt.severity_level
ORDER BY COUNT(se.id) DESC

-- Evenements recents avec details (DOUBLE JOIN)

SELECT se.*, s.name AS server_name, tt.name AS threat_name
FROM security_events se
LEFT JOIN servers s ON se.server_id = s.id
LEFT JOIN threat_types tt ON se.threat_type_id = tt.id
ORDER BY se.event_timestamp DESC

-- Metriques reseau moyennes par serveur (JOIN + AVG)

SELECT s.name, AVG(nm.bandwidth_in), AVG(nm.latency_ms)
FROM servers s
LEFT JOIN network_metrics nm ON s.id = nm.server_id
GROUP BY s.id, s.name
HAVING COUNT(nm.id) > 0
